let multiplier = 1; let timer; let running = false; const rocket = document.getElementById("rocket"); const display = document.getElementById("multiplier"); function startGame(){ if(running) return; running=true; multiplier=1; timer=setInterval(()=>{ multiplier += 0.05; display.innerHTML = multiplier.toFixed(2)+"x"; rocket.style.bottom = (100 + multiplier*20)+"px"; rocket.style.left = (50 + multiplier)+"%"; if(multiplier >= 10){ clearInterval(timer); display.innerHTML="💥 CRASH"; running=false; } },100); } function cashOut(){ if(running){ alert( "WIN 💎 "+multiplier.toFixed(2)+"x" ); }else{ alert("بازی شروع نشده"); } }
