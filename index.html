import express from "express"; import dotenv from "dotenv"; import crypto from "crypto"; import Database from "better-sqlite3"; import path from "path"; import { fileURLToPath } from "url"; dotenv.config(); const __filename = fileURLToPath(import.meta.url); const __dirname = path.dirname(__filename); const BOT_TOKEN = process.env.BOT_TOKEN; const ADMIN_ID = String(process.env.ADMIN_ID || "7681716638"); const PORT = Number(process.env.PORT || 3000); if (!BOT_TOKEN) { console.error("BOT_TOKEN داخل .env تنظیم نشده!"); process.exit(1); } const db = new Database(path.join(__dirname, "data.sqlite")); db.pragma("journal_mode = WAL"); db.exec(` CREATE TABLE IF NOT EXISTS users ( telegram_id TEXT PRIMARY KEY, username TEXT DEFAULT '', first_name TEXT DEFAULT '', balance INTEGER NOT NULL DEFAULT 50, created_at TEXT DEFAULT CURRENT_TIMESTAMP, updated_at TEXT DEFAULT CURRENT_TIMESTAMP ); CREATE TABLE IF NOT EXISTS games ( id INTEGER PRIMARY KEY AUTOINCREMENT, telegram_id TEXT NOT NULL, bet INTEGER NOT NULL, crash_at REAL NOT NULL, status TEXT NOT NULL DEFAULT 'running', payout INTEGER NOT NULL DEFAULT 0, multiplier REAL DEFAULT 1, started_at INTEGER NOT NULL, finished_at INTEGER ); `); const app = express(); app.use(express.json()); app.use(express.static(path.join(__dirname, "public"))); function verifyTelegram(initData) { if (!initData) return null; const params = new URLSearchParams(initData); const hash = params.get("hash"); if (!hash) return null; params.delete("hash"); const dataCheckString = [...params.entries()] .sort(([a], [b]) => a.localeCompare(b)) .map(([key, value]) => `${key}=${value}`) .join("\n"); const secretKey = crypto .createHmac("sha256", "WebAppData") .update(BOT_TOKEN) .digest(); const calculatedHash = crypto .createHmac("sha256", secretKey) .update(dataCheckString) .digest("hex"); if (calculatedHash !== hash) return null; const authDate = Number(params.get("auth_date")); if (!authDate) return null; if (Math.floor(Date.now() / 1000) - authDate > 86400) { return null; } try { const user = JSON.parse(params.get("user") || "{}"); if (!user.id) return null; return user; } catch { return null; } } function auth(req, res, next) { const user = verifyTelegram( req.headers["x-telegram-init-data"] ); if (!user) { return res.status(401).json({ ok: false, error: "Telegram authentication failed" }); } req.telegramUser = user; next(); } function createUser(user) { db.prepare(` INSERT INTO users (telegram_id, username, first_name) VALUES (?, ?, ?) ON CONFLICT(telegram_id) DO UPDATE SET username = excluded.username, first_name = excluded.first_name, updated_at = CURRENT_TIMESTAMP `).run( String(user.id), user.username || "", user.first_name || "" ); } function getUser(id) { return db .prepare("SELECT * FROM users WHERE telegram_id = ?") .get(String(id)); } function getActiveGame(id) { return db .prepare(` SELECT * FROM games WHERE telegram_id = ? AND status = 'running' ORDER BY id DESC LIMIT 1 `) .get(String(id)); } function crashPoint() { const r = Math.random(); return Number( Math.max( 1.05, Math.min( 30, 1 + Math.pow(1 / Math.max(0.05, 1 - r), 1.15) * 0.55 ) ).toFixed(2) ); } /* اطلاعات کاربر */ app.get("/api/me", auth, (req, res) => { createUser(req.telegramUser); const user = getUser(req.telegramUser.id); res.json({ ok: true, user: { id: String(req.telegramUser.id), username: req.telegramUser.username || "", firstName: req.telegramUser.first_name || "", balance: user.balance, isAdmin: String(req.telegramUser.id) === ADMIN_ID } }); }); /* پنل ادمین */ app.post("/api/admin/balance", auth, (req, res) => { if (String(req.telegramUser.id) !== ADMIN_ID) { return res.status(403).json({ ok: false, error: "دسترسی ادمین ندارید" }); } const userId = String(req.body.userId || "").trim(); const amount = Math.floor(Number(req.body.amount)); const mode = req.body.mode === "remove" ? "remove" : "add"; if (!/^\d+$/.test(userId)) { return res.status(400).json({ ok: false, error: "Telegram ID نامعتبر است" }); } if (!Number.isFinite(amount) || amount < 1) { return res.status(400).json({ ok: false, error: "تعداد توکن نامعتبر است" }); } let user = getUser(userId); if (!user) { db.prepare(` INSERT INTO users (telegram_id, balance) VALUES (?, 0) `).run(userId); user = getUser(userId); } let newBalance; if (mode === "add") { newBalance = user.balance + amount; } else { newBalance = Math.max(0, user.balance - amount); } db.prepare(` UPDATE users SET balance = ?, updated_at = CURRENT_TIMESTAMP WHERE telegram_id = ? `).run(newBalance, userId); res.json({ ok: true, userId, balance: newBalance }); }); /* شروع بازی */ app.post("/api/game/start", auth, (req, res) => { const id = String(req.telegramUser.id); createUser(req.telegramUser); const bet = Math.floor(Number(req.body.bet)); if (!Number.isFinite(bet) || bet < 1) { return res.status(400).json({ ok: false, error: "مبلغ شرط نامعتبر است" }); } if (getActiveGame(id)) { return res.status(409).json({ ok: false, error: "یک بازی در حال اجراست" }); } const user = getUser(id); if (user.balance < bet) { return res.status(400).json({ ok: false, error: "توکن کافی نیست" }); } const crashAt = crashPoint(); const result = db.transaction(() => { db.prepare(` UPDATE users SET balance = balance - ? WHERE telegram_id = ? `).run(bet, id); const game = db.prepare(` INSERT INTO games (telegram_id, bet, crash_at, started_at) VALUES (?, ?, ?, ?) `).run( id, bet, crashAt, Date.now() ); return game.lastInsertRowid; })(); res.json({ ok: true, gameId: result, bet, crashAt, balance: getUser(id).balance }); }); /* وضعیت بازی */ app.get("/api/game/state", auth, (req, res) => { const id = String(req.telegramUser.id); const game = getActiveGame(id); if (!game) { return res.json({ ok: true, running: false, balance: getUser(id).balance }); } const seconds = (Date.now() - game.started_at) / 1000; const multiplier = Number( (1 + Math.pow(seconds, 1.45) * 0.34).toFixed(2) ); if (multiplier >= game.crash_at) { db.prepare(` UPDATE games SET status = 'crashed', multiplier = ?, finished_at = ? WHERE id = ? AND status = 'running' `).run( game.crash_at, Date.now(), game.id ); return res.json({ ok: true, running: false, crashed: true, multiplier: game.crash_at, balance: getUser(id).balance }); } res.json({ ok: true, running: true, multiplier, balance: getUser(id).balance }); }); /* برداشت */ app.post("/api/game/cashout", auth, (req, res) => { const id = String(req.telegramUser.id); const game = getActiveGame(id); if (!game) { return res.status(400).json({ ok: false, error: "بازی فعالی وجود ندارد" }); } const seconds = (Date.now() - game.started_at) / 1000; const multiplier = Number( (1 + Math.pow(seconds, 1.45) * 0.34).toFixed(2) ); if (multiplier >= game.crash_at) { db.prepare(` UPDATE games SET status = 'crashed', multiplier = ?, finished_at = ? WHERE id = ? AND status = 'running' `).run( game.crash_at, Date.now(), game.id ); return res.json({ ok: true, crashed: true, multiplier: game.crash_at, balance: getUser(id).balance }); } const payout = Math.floor( game.bet * multiplier ); db.transaction(() => { db.prepare(` UPDATE games SET status = 'cashed_out', multiplier = ?, payout = ?, finished_at = ? WHERE id = ? AND status = 'running' `).run( multiplier, payout, Date.now(), game.id ); db.prepare(` UPDATE users SET balance = balance + ? WHERE telegram_id = ? `).run( payout, id ); })(); res.json({ ok: true, crashed: false, multiplier, payout, balance: getUser(id).balance }); }); app.get("*", (req, res) => { res.sendFile( path.join(__dirname, "public", "index.html") ); }); app.listen(PORT, () => { console.log(`Server running on port ${PORT}`); });
